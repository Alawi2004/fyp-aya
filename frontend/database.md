users (1) ─── (1) drivers
•  Every driver is a user.
•  A user can be a driver only once.
•  Drivers log in like normal users but have extra driver-specific data (license, hire date).
 
users (1) ─── (M) tickets
•  A user can buy many tickets.
•  Each ticket belongs to exactly one user.

• A user receives many notifications.
• Each notification belongs to one user.
Example notifications:
• Bus arriving
• Delay alert
• Ticket confirmation
 
users (1) ─── (M) ratings
•  A user can submit many ratings.
•  Each rating belongs to one user.
 
vehicles (1) ─── (M) trips
•  One vehicle can perform many trips.
•  Each trip uses exactly one vehicle.
 
drivers (1) ─── (M) trips
•  One driver can drive many trips.
•  Each trip has exactly one driver.
 
routes (1) ─── (M) trips
•  A route can be used for many trips.
•  Each trip follows one route.
 
routes (M) ─── (M) stops
via route_stops
•  One route has many stops.
•  A stop can belong to many routes.

1. Users & Drivers
• users
Stores all system users with identity and authentication info (name, email, password, role).
Roles include passengers, drivers, and admins.
• drivers
Extends users who are drivers by adding driver-specific details (license number, hire date).
Each driver is linked to exactly one user. A user can be either a driver or not.
 
2. Vehicles
• vehicles
Stores all vehicles available in the system with details like plate number, type (bus, van, car, shuttle), capacity, and status (active, maintenance, inactive).
Vehicles are assigned to trips.
 
3. Routes & Stops
• routes
Defines bus routes with a name and start/end locations.
• stops
Lists all stops with name and GPS coordinates (latitude, longitude).
• route_stops (junction table)
Connects many stops to many routes and defines the order of stops in each route.
Enables routes to share stops and stops to be part of multiple routes.
 
4. Trips
• trips
Represents a single trip made by a vehicle and driver on a route, with start and end times and status (scheduled, ongoing, completed, cancelled).
Links to a vehicle, driver, and route.
 
5. Logs & Measurements for Trips
• gps_logs
Stores frequent GPS position records (latitude, longitude, timestamp) generated by the vehicle during a trip.
• passenger_counts
Records passenger counts captured during a trip by AI cameras or QR scans, including method used.
• eta_predictions
Stores estimated time of arrival predictions made over the course of a trip with confidence scores.
 
6. Tickets & Passenger Interaction
• tickets
Tickets booked by users for specific trips with seat numbers and booking timestamps.
• notifications
Messages sent to users related to the system, such as bus arrivals, delay alerts, or ticket confirmations. Includes read/unread status.
• ratings
Ratings and optional comments submitted by users for trips they took, with ratings constrained to values between 1 and 5.
 
7. Relationships Summary
• One user can be one driver (1-to-1).
• One user can have many tickets, notifications, and ratings (1-to-many).
• One vehicle can be assigned to many trips (1-to-many).
• One driver can drive many trips (1-to-many).
• One route can have many trips (1-to-many).
• Many routes have many stops via route_stops (many-to-many).
• One trip has many gps_logs, passenger_counts, tickets, ratings, and eta_predictions (1-to-many).

CREATE TABLE users (
   user_id INT IDENTITY(1,1) PRIMARY KEY,
   full_name VARCHAR(100) NOT NULL,
   email VARCHAR(120) UNIQUE NOT NULL,
   password_hash VARCHAR(255) NOT NULL,
   phone VARCHAR(20),
   role VARCHAR(20) DEFAULT 'passenger',
   created_at DATETIME DEFAULT GETDATE()
);
 
ALTER TABLE users
ADD CONSTRAINT chk_role
CHECK (role IN ('passenger', 'driver', 'admin'));
 
CREATE TABLE vehicles (
   vehicle_id INT IDENTITY(1,1) PRIMARY KEY,
   plate_number VARCHAR(20) NOT NULL UNIQUE,
   vehicle_type VARCHAR(20) NOT NULL,
   capacity INT NOT NULL,
   model VARCHAR(50),
   status VARCHAR(20) DEFAULT 'active',
   created_at DATETIME DEFAULT GETDATE()
);
ALTER TABLE vehicles
ADD CONSTRAINT chk_vehicle_type
CHECK (vehicle_type IN ('bus','van','car','shuttle'));
 
ALTER TABLE vehicles
ADD CONSTRAINT chk_vehicle_status
CHECK (status IN ('active','maintenance','inactive'));
 
CREATE TABLE drivers (
   driver_id INT IDENTITY(1,1) PRIMARY KEY,
   user_id INT NOT NULL UNIQUE,
   license_number VARCHAR(50) NOT NULL,
   hire_date DATE,
 
   CONSTRAINT fk_drivers_users FOREIGN KEY (user_id) REFERENCES users(user_id)
);
 
CREATE TABLE routes (
   route_id INT IDENTITY(1,1) PRIMARY KEY,
   route_name VARCHAR(100) NOT NULL,
   start_location VARCHAR(100),
   end_location VARCHAR(100)
);
 
CREATE TABLE stops (
   stop_id INT IDENTITY(1,1) PRIMARY KEY,
   stop_name VARCHAR(100) NOT NULL,
   latitude DECIMAL(9,6),
   longitude DECIMAL(9,6)
);
 
CREATE TABLE route_stops (
   route_id INT,
   stop_id INT,
   stop_order INT,
 
   CONSTRAINT pk_route_stops PRIMARY KEY (route_id, stop_id),
   CONSTRAINT fk_route_stops_routes FOREIGN KEY (route_id) REFERENCES routes(route_id),
   CONSTRAINT fk_route_stops_stops FOREIGN KEY (stop_id) REFERENCES stops(stop_id)
);
 
CREATE TABLE trips (
   trip_id INT IDENTITY(1,1) PRIMARY KEY,
   vehicle_id INT NOT NULL,
   driver_id INT NOT NULL,
   route_id INT NOT NULL,
   start_time DATETIME,
   end_time DATETIME,
   status VARCHAR(20),
 
   CONSTRAINT fk_trips_vehicles FOREIGN KEY (vehicle_id) REFERENCES vehicles(vehicle_id),
   CONSTRAINT fk_trips_drivers FOREIGN KEY (driver_id) REFERENCES drivers(driver_id),
   CONSTRAINT fk_trips_routes FOREIGN KEY (route_id) REFERENCES routes(route_id)
);
 
ALTER TABLE trips
ADD CONSTRAINT chk_trip_status
CHECK (status IN ('scheduled', 'ongoing', 'completed', 'cancelled'));
 
CREATE TABLE gps_logs (
   gps_id INT IDENTITY(1,1) PRIMARY KEY,
   trip_id INT NOT NULL,
   latitude DECIMAL(9,6),
   longitude DECIMAL(9,6),
   recorded_at DATETIME DEFAULT GETDATE(),
 
   CONSTRAINT fk_gpslogs_trips FOREIGN KEY (trip_id) REFERENCES trips(trip_id)
);
 
CREATE TABLE passenger_counts (
   count_id INT IDENTITY(1,1) PRIMARY KEY,
   trip_id INT NOT NULL,
   passenger_count INT NOT NULL,
   recorded_at DATETIME DEFAULT GETDATE(),
   method VARCHAR(20) NOT NULL DEFAULT 'camera',
 
   CONSTRAINT fk_passengercounts_trips FOREIGN KEY (trip_id) REFERENCES trips(trip_id),
   CONSTRAINT chk_method CHECK (method IN ('camera', 'qr_scan'))
);
 
CREATE TABLE tickets (
   ticket_id INT IDENTITY(1,1) PRIMARY KEY,
   user_id INT NOT NULL,
   trip_id INT NOT NULL,
   seat_number VARCHAR(10),
   booking_time DATETIME DEFAULT GETDATE(),
 
   CONSTRAINT fk_tickets_users FOREIGN KEY (user_id) REFERENCES users(user_id),
   CONSTRAINT fk_tickets_trips FOREIGN KEY (trip_id) REFERENCES trips(trip_id)
);
 
CREATE TABLE notifications (
   notification_id INT IDENTITY(1,1) PRIMARY KEY,
   user_id INT NOT NULL,
   message TEXT NOT NULL,
   is_read BIT DEFAULT 0,
   created_at DATETIME DEFAULT GETDATE(),
 
   CONSTRAINT fk_notifications_users FOREIGN KEY (user_id) REFERENCES users(user_id)
);
 
CREATE TABLE eta_predictions (
   eta_id INT IDENTITY(1,1) PRIMARY KEY,
   trip_id INT NOT NULL,
   predicted_arrival DATETIME,
   confidence DECIMAL(5,2),
 
   CONSTRAINT fk_eta_predictions_trips FOREIGN KEY (trip_id) REFERENCES trips(trip_id)
);
 
CREATE TABLE ratings (
   rating_id INT IDENTITY(1,1) PRIMARY KEY,
   user_id INT NOT NULL,
   trip_id INT NOT NULL,
   rating INT CHECK (rating BETWEEN 1 AND 5),
   comment TEXT,
 
   CONSTRAINT fk_ratings_users FOREIGN KEY (user_id) REFERENCES users(user_id),
   CONSTRAINT fk_ratings_trips FOREIGN KEY (trip_id) REFERENCES trips(trip_id)
);
 
CREATE VIEW view_trip_vehicle AS
SELECT
   t.trip_id,
   v.vehicle_type,
   v.plate_number,
   v.capacity,
   t.status,
   r.route_name
FROM trips t
JOIN vehicles v ON t.vehicle_id = v.vehicle_id
JOIN routes r ON t.route_id = r.route_id;
 
CREATE VIEW view_passenger_load AS
SELECT
   t.trip_id,
   MAX(pc.passenger_count) AS current_passengers,
   v.capacity
FROM trips t
JOIN passenger_counts pc ON t.trip_id = pc.trip_id
JOIN vehicles v ON t.vehicle_id = v.vehicle_id
GROUP BY t.trip_id, v.capacity;
 
CREATE PROCEDURE book_ticket
   @p_user_id INT,
   @p_trip_id INT,
   @p_seat VARCHAR(10)
AS
BEGIN
   INSERT INTO tickets (user_id, trip_id, seat_number)
   VALUES (@p_user_id, @p_trip_id, @p_seat);
END;
 
CREATE PROCEDURE get_trips_by_vehicle
   @p_type VARCHAR(20)
AS
BEGIN
   SELECT *
   FROM view_trip_vehicle
   WHERE vehicle_type = @p_type;
END;
